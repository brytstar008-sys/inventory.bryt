<script>

// INVENTORY STORAGE

let records = JSON.parse(
    localStorage.getItem("courtInventory")
) || [];

let editingIndex = null;


// GET FORM VALUES

function getValue(id) {
    return Math.max(0, Number(document.getElementById(id).value) || 0);
}


// SAVE OR UPDATE

document.getElementById("inventoryForm")
.addEventListener("submit", function(e) {

    e.preventDefault();

    const record = {

        court: document.getElementById("court").value.trim(),
        room: document.getElementById("room").value.trim(),
        date: document.getElementById("date").value,

        suW: getValue("suW"),
        suF: getValue("suF"),

        prW: getValue("prW"),
        prF: getValue("prF"),

        moW: getValue("moW"),
        moF: getValue("moF"),

        laW: getValue("laW"),
        laF: getValue("laF")

    };

    const duplicate = records.some((r, i) =>
        r.court.toLowerCase() === record.court.toLowerCase() &&
        r.room.toLowerCase() === record.room.toLowerCase() &&
        i !== editingIndex
    );

    if (duplicate) {
        alert("This courtroom already has an inventory record. Edit the existing record instead.");
        return;
    }

    if (editingIndex !== null) {
        records[editingIndex] = record;
    } else {
        records.push(record);
    }

    saveData();
    this.reset();
    cancelEdit();
    displayRecords();

});


// SAVE DATA

function saveData() {
    localStorage.setItem("courtInventory", JSON.stringify(records));
}


// CALCULATE TOTALS

function calculate(r) {

    let working = r.suW + r.prW + r.moW + r.laW;

    let faulty = r.suF + r.prF + r.moF + r.laF;

    return {
        working: working,
        faulty: faulty,
        total: working + faulty,

        suTotal: r.suW + r.suF,
        prTotal: r.prW + r.prF,
        moTotal: r.moW + r.moF,
        laTotal: r.laW + r.laF
    };

}


// DISPLAY RECORDS

function displayRecords() {

    const tbody = document.getElementById("inventoryBody");

    tbody.innerHTML = "";

    const query = document.getElementById("search")
    .value.toLowerCase();

    let filtered = records.map((r,index) => ({
        ...r,
        originalIndex: index
    })).filter(r =>
        r.court.toLowerCase().includes(query) ||
        r.room.toLowerCase().includes(query)
    );

    let grand = {
        suW:0,suF:0,suTotal:0,
        prW:0,prF:0,prTotal:0,
        moW:0,moF:0,moTotal:0,
        laW:0,laF:0,laTotal:0,
        working:0,faulty:0,total:0
    };

    filtered.forEach((r,index) => {

        const t = calculate(r);

        let values = [
            index+1,
            r.court,
            r.room,

            r.suW,r.suF,t.suTotal,
            r.prW,r.prF,t.prTotal,
            r.moW,r.moF,t.moTotal,
            r.laW,r.laF,t.laTotal,

            t.total,t.working,t.faulty,r.date
        ];

        const tr = document.createElement("tr");

        values.forEach((value,i) => {

            const td = document.createElement("td");
            td.textContent = value;

            if ([3,6,9,12,16].includes(i)) {
                td.className = "working";
            }

            if ([4,7,10,13,17].includes(i)) {
                td.className = "not-working";
            }

            tr.appendChild(td);

        });

        const actions = document.createElement("td");
        actions.className = "action-column";

        const edit = document.createElement("button");
        edit.textContent = "Edit";
        edit.className = "edit-btn";
        edit.onclick = () => editRecord(r.originalIndex);

        const del = document.createElement("button");
        del.textContent = "Delete";
        del.className = "delete-btn";
        del.onclick = () => deleteRecord(r.originalIndex);

        actions.appendChild(edit);
        actions.appendChild(del);

        tr.appendChild(actions);

        tbody.appendChild(tr);

        // GRAND TOTALS

        grand.suW += r.suW;
        grand.suF += r.suF;
        grand.suTotal += t.suTotal;

        grand.prW += r.prW;
        grand.prF += r.prF;
        grand.prTotal += t.prTotal;

        grand.moW += r.moW;
        grand.moF += r.moF;
        grand.moTotal += t.moTotal;

        grand.laW += r.laW;
        grand.laF += r.laF;
        grand.laTotal += t.laTotal;

        grand.working += t.working;
        grand.faulty += t.faulty;
        grand.total += t.total;

    });

    // TOTAL ROW

    const footer = document.getElementById("tableFooter");
    footer.innerHTML = "";

    const totalRow = document.createElement("tr");
    totalRow.className = "total-row";

    const totalValues = [
        "", "GRAND TOTAL", "",
        grand.suW,grand.suF,grand.suTotal,
        grand.prW,grand.prF,grand.prTotal,
        grand.moW,grand.moF,grand.moTotal,
        grand.laW,grand.laF,grand.laTotal,
        grand.total,grand.working,grand.faulty,
        "",""
    ];

    totalValues.forEach(value => {
        const td = document.createElement("td");
        td.textContent = value;
        totalRow.appendChild(td);
    });

    footer.appendChild(totalRow);

    updateDashboard();

}


// DASHBOARD TOTALS

function updateDashboard() {

    let totalWorking = 0;
    let totalFaulty = 0;
    let totalEquipment = 0;

    records.forEach(r => {

        const t = calculate(r);

        totalWorking += t.working;
        totalFaulty += t.faulty;
        totalEquipment += t.total;

    });

    document.getElementById("courtTotal").textContent =
    records.length;

    document.getElementById("equipmentTotal").textContent =
    totalEquipment;

    document.getElementById("workingTotal").textContent =
    totalWorking;

    document.getElementById("faultyTotal").textContent =
    totalFaulty;

}


// EDIT RECORD

function editRecord(index) {

    const r = records[index];

    document.getElementById("court").value = r.court;
    document.getElementById("room").value = r.room;
    document.getElementById("date").value = r.date;

    ["suW","suF","prW","prF","moW","moF","laW","laF"]
    .forEach(id => {
        document.getElementById(id).value = r[id];
    });

    editingIndex = index;

    document.getElementById("formTitle").textContent =
    "Edit Courtroom Inventory";

    document.getElementById("saveBtn").textContent =
    "Update Inventory";

    document.getElementById("cancelBtn").style.display =
    "inline-block";

    document.getElementById("inventoryForm")
    .scrollIntoView({behavior:"smooth"});

}


// CANCEL EDIT

function cancelEdit() {

    editingIndex = null;

    document.getElementById("inventoryForm").reset();

    document.getElementById("formTitle").textContent =
    "Add Courtroom Inventory";

    document.getElementById("saveBtn").textContent =
    "+ Save Courtroom Inventory";

    document.getElementById("cancelBtn").style.display =
    "none";

}


// DELETE RECORD

function deleteRecord(index) {

    if(confirm("Delete this courtroom inventory record?")) {

        records.splice(index,1);

        saveData();
        displayRecords();

    }

}


// SEARCH

document.getElementById("search")
.addEventListener("input", displayRecords);


// PRINT

function printInventory() {

    window.print();

}


// INITIAL DISPLAY

displayRecords();

</script>
